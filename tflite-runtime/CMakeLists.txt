cmake_minimum_required(VERSION 3.22)

project(tflite-runtime)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(CMAKE_VERSION VERSION_GREATER_EQUAL "3.24")
	cmake_policy(SET CMP0135 NEW)
endif()

option(TFLITE_RUNTIME_USE_PREBUILT_TFLITE "Use prebuilt TFLite library from LiteRT in third_party/litert(-gpu)-1.2.0 (requires TFLITE_RUNTIME_ABI)" OFF)

option(TFLITE_RUNTIME_ENABLE_ASAN "Enable AddressSanitizer" OFF)

option(TFLITE_RUNTIME_ENABLE_CLANG_TIDY "Enable Clang Tidy" OFF)

option(TFLITE_RUNTIME_GENERATE_DOCS "Generate docs using doxygen" OFF)

option(TFLITE_RUNTIME_ENABLE_TRACY_PROFILER "Enable tracy profiler" OFF)

if (TFLITE_RUNTIME_ENABLE_CLANG_TIDY)
	find_program(CLANG_TIDY_EXE NAMES "clang-tidy")
	if (${CLANG_TIDY_EXE} STREQUAL "CLANG_TIDY_EXE-NOTFOUND")
		message(WARNING "clang-tidy not found, not using clang-tidy for static analysis despite TFLITE_RUNTIME_ENABLE_CLANG_TIDY being ON")
		unset(CLANG_TIDY_EXE CACHE)
	endif ()
endif()

find_program(CCACHE_EXE ccache)
if(NOT (${CCACHE_EXE} STREQUAL "CCACHE_EXE-NOTFOUND"))
	set_property(GLOBAL PROPERTY RULE_LAUNCH_COMPILE ccache)
	set_property(GLOBAL PROPERTY RULE_LAUNCH_LINK ccache)
endif()

if (DEFINED CMAKE_ANDROID_ARCH_ABI)
	set(TFLITE_RUNTIME_USE_PREBUILT_TFLITE ON CACHE BOOL "Use prebuilt TFLite library from LiteRT in third_party/litert(-gpu)-1.2.0 (requires TFLITE_RUNTIME_ABI)" FORCE)
	set(TFLITE_RUNTIME_ABI ${CMAKE_ANDROID_ARCH_ABI})
	set(TFLITE_RUNTIME_FORCE_OBOE "Force OpenAL to use oboe backend (for android)" ON)
else ()
	message(STATUS "tflite-runtime: Not using Android NDK, setting tflite-runtime ABI to host arch")
	if (CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64")
		set(TFLITE_RUNTIME_ABI "x86_64")
	elseif (CMAKE_SYSTEM_PROCESSOR MATCHES "i386")
		set(TFLITE_RUNTIME_ABI "x86")
	elseif (CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
		set(TFLITE_RUNTIME_ABI "arm64-v8a")
	else ()
		set(TFLITE_RUNTIME_ABI "armeabi-v7a")
	endif ()
endif ()

message(STATUS "tflite-runtime ABI: ${TFLITE_RUNTIME_ABI}")

if (TFLITE_RUNTIME_ENABLE_ASAN)
	message(STATUS "Enabling AddressSanitizer")
	add_compile_options(-fsanitize=address,undefined,leak -fno-omit-frame-pointer -fno-optimize-sibling-calls)
	add_link_options(-fsanitize=address,undefined,leak -static-libasan)
endif()

add_subdirectory(third_party)

file(GLOB_RECURSE TFLITE_RUNTIME_SOURCE_FILES
	"${CMAKE_CURRENT_SOURCE_DIR}/include/tflite-runtime/*.hpp"
	"${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
)

if (TFLITE_RUNTIME_USE_PREBUILT_TFLITE)
	add_library(
		tflite-runtime
		STATIC
		${TFLITE_RUNTIME_SOURCE_FILES}
	)
else()
	add_library(
		tflite-runtime
		SHARED
		${TFLITE_RUNTIME_SOURCE_FILES}
	)
endif()

target_compile_features(tflite-runtime PUBLIC cxx_std_20)
target_compile_options(tflite-runtime PRIVATE
	$<$<OR:$<CXX_COMPILER_ID:Clang>,$<CXX_COMPILER_ID:AppleClang>,$<CXX_COMPILER_ID:GNU>>:
		-Wall -Wextra -Werror -Wno-c99-extensions>
	$<$<CXX_COMPILER_ID:MSVC>:
		/W4 /WX>
)

set_property(TARGET tflite-runtime PROPERTY COMPILE_WARNING_AS_ERROR ON)

target_include_directories(tflite-runtime
	PUBLIC
	"${CMAKE_CURRENT_SOURCE_DIR}/include"
	${THIRD_PARTY_INCLUDE_DIRS}
)

target_link_libraries(
	tflite-runtime
	PUBLIC
	${THIRD_PARTY_LIBS}
)

# --- Generate link manifest for Rust build.rs ---
# This file lists all dynamic library dependencies so that the Rust
# build script can forward them as cargo:rustc-link-lib / cargo:rustc-link-search
# directives without hardcoding any paths or library names.
set(_tflite_manifest "${CMAKE_BINARY_DIR}/tflite-cargo-link.txt")
file(WRITE "${_tflite_manifest}" "# Auto-generated by CMake, DO NOT EDIT!\n")
# Static lib (tflite-runtime itself)
file(APPEND "${_tflite_manifest}"
	"cargo:rustc-link-search=native=${CMAKE_BINARY_DIR}\n")
if (TFLITE_RUNTIME_USE_PREBUILT_TFLITE)
	file(APPEND "${_tflite_manifest}"
		"cargo:rustc-link-lib=static=tflite-runtime\n"
	)
else()
	file(APPEND "${_tflite_manifest}"
		"cargo:rustc-link-lib=dylib=tflite-runtime\n"
	)
endif()
# Dynamic dependencies from THIRD_PARTY_LIBS
foreach(_lib ${THIRD_PARTY_LIBS})
	# Skip CMake targets (tl::expected, tensorflowlite_c, absl::log, etc.)
	# — they are either header-only or static libs that are already linked
	#   into tflite-runtime when building from source.
	if(IS_ABSOLUTE "${_lib}" AND EXISTS "${_lib}")
		get_filename_component(_dir "${_lib}" DIRECTORY)
		get_filename_component(_name "${_lib}" NAME_WE)
		# Strip "lib" prefix -> "libfoo_jni.so" becomes "foo_jni"
		string(REGEX REPLACE "^lib" "" _name "${_name}")
		file(APPEND "${_tflite_manifest}"
			"cargo:rustc-link-search=native=${_dir}\n"
			"cargo:rustc-link-lib=dylib=${_name}\n"
		)
	endif()
endforeach()
message(STATUS "tflite-runtime: wrote link manifest to ${_tflite_manifest}")

if(TFLITE_RUNTIME_USE_PREBUILT_TFLITE)
	target_compile_definitions(tflite-runtime PUBLIC TFLITE_RUNTIME_USE_PREBUILT_TFLITE=1)
else()
	target_compile_definitions(tflite-runtime PUBLIC TFLITE_RUNTIME_USE_PREBUILT_TFLITE=0)
endif()

if (TFLITE_RUNTIME_ENABLE_CLANG_TIDY AND (DEFINED CLANG_TIDY_EXE))
	set_target_properties(tflite-runtime PROPERTIES CXX_CLANG_TIDY "${CLANG_TIDY_EXE}")
endif()

if (TFLITE_RUNTIME_ENABLE_TRACY_PROFILER)
	target_link_libraries(tflite-runtime PUBLIC Tracy::TracyClient)
	target_compile_definitions(tflite-runtime PUBLIC TFLITE_RUNTIME_ENABLE_TRACY_PROFILER=1)
else()
	target_compile_definitions(tflite-runtime PUBLIC TFLITE_RUNTIME_ENABLE_TRACY_PROFILER=0)
endif()
